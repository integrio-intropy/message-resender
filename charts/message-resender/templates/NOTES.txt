To create a ResendPolicy:

  cat <<EOF | kubectl apply -f -
  apiVersion: messaging.intropy.io/v1
  kind: ResendPolicy
  metadata:
    name: my-queue
    namespace: {{ .Release.Namespace }}
  spec:
    queueName: my-queue
  EOF
